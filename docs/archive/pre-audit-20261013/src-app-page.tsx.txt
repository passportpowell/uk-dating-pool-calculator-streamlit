"use client";
import { useEffect, useMemo, useState } from "react";
import Link from "next/link";
import SiteShell from "@/components/SiteShell";
import { DEFAULT_CRITERIA, validateCriteria, calculatePool, geographies, sourceLibrary, childrenStatus, EDUCATION_LABELS, RELATIONSHIP_LABELS, ORIENTATION_LABELS, BMI_LABELS, ETHNICITY_LABELS, SEX_LABELS, CHILDREN_LABELS, INCOME_OPTIONS, type Criteria } from "@/lib/calculator";

const people = (n: number) => n === 0 ? "0" : n < 1 ? "less than 1" : new Intl.NumberFormat("en-GB", {maximumSignificantDigits: 2}).format(n);
const pct = (n: number) => n === 0 ? "0%" : n < .00001 ? "<0.00001%" : new Intl.NumberFormat("en-GB", {maximumSignificantDigits: 3}).format(n) + "%";
function Select({label,value,options,onChange,disabled=false}:{label:string,value:string,options:Record<string,string>,onChange:(v:string)=>void,disabled?:boolean}) { return <label className="field">{label}<select value={value} onChange={e=>onChange(e.target.value)} disabled={disabled}>{Object.entries(options).map(([key,text])=><option key={key} value={key}>{text}</option>)}</select></label>; }
function HeightBound({label,value,unit,onChange}:{label:string,value:number|null,unit:string,onChange:(v:number|null)=>void}) {
  const inches = value === null || !Number.isFinite(value) ? null : Math.round(value/2.54*100)/100;
  const feet = inches === null ? "" : Math.floor(inches/12);
  const inch = inches === null ? "" : Number((inches % 12).toFixed(2));
  return unit === "cm" ? <label>{label}, cm<input type="number" min="100" max="250" step="0.01" placeholder="Any" value={value!==null&&Number.isFinite(value)?value:""} onChange={e=>onChange(e.target.value===""?null:Number(e.target.value))}/></label> : <div><span className="small">{label}</span><div className="pair"><label>{label} feet<input type="number" min="3" max="8" step="1" placeholder="Any" value={feet} onChange={e=>{const n=Number(e.target.value);onChange(e.target.value===""?null:!Number.isInteger(n)?NaN:(n*12+Number(inch))*2.54);}}/></label><label>{label} inches<input type="number" min="0" max="11.99" step=".01" placeholder="Any" value={inch} onChange={e=>{const n=Number(e.target.value);onChange(e.target.value===""?null:n<0||n>=12?NaN:(Number(feet)*12+n)*2.54);}}/></label></div></div>;
}
export default function Calculator() {
  const [criteria,setCriteria]=useState<Criteria>(DEFAULT_CRITERIA);
  const [unit,setUnit]=useState("cm");
  const [ready,setReady]=useState(false);
  const [message,setMessage]=useState("");
  const [linkError,setLinkError]=useState("");
  const [scenarios,setScenarios]=useState<Criteria[]>([]);
  const update=(patch:Partial<Criteria>)=>{setCriteria(c=>({...c,...patch}));setMessage("");};
  useEffect(()=>{
    const restore=()=>{try{const raw=new URL(window.location.href).searchParams.get("c");if(raw){const payload=JSON.parse(raw);if(payload.v!==1)throw new Error("Unsupported link version.");setCriteria(validateCriteria(payload.criteria));}else setCriteria(DEFAULT_CRITERIA);setLinkError("");}catch(e){setLinkError("This shared link could not be restored. "+(e instanceof Error?e.message:"Invalid criteria."));}setReady(true);};
    queueMicrotask(restore);window.addEventListener("popstate",restore);return()=>window.removeEventListener("popstate",restore);
  },[]);
  const calculation=useMemo(()=>{try{return {result:calculatePool(validateCriteria(criteria)),error:""};}catch(e){return {result:null,error:e instanceof Error?e.message:"Check your requirements."};}},[criteria]);
  const result=calculation.result;
  useEffect(()=>{if(!ready||!result)return;const url=new URL(window.location.href);url.searchParams.set("c",JSON.stringify({v:1,criteria:result.state}));window.history.replaceState(null,"",url);},[ready,result]);
  const copy=async()=>{try{await navigator.clipboard.writeText(window.location.href);setMessage("Link copied. All requirements are included.");}catch{setMessage("Copy the address from your browser to share these requirements.");}};
  const download=()=>{if(!result)return;const blob=new Blob([JSON.stringify({version:1,criteria:result.state,count:result.count,base:result.base,p:result.p,stages:result.stages,notes:result.notes,sources:sourceLibrary},null,2)],{type:"application/json"});const url=URL.createObjectURL(blob);const a=document.createElement("a");a.href=url;a.download="dating-pool-result.json";a.click();URL.revokeObjectURL(url);setMessage("Result downloaded with requirements and sources.");};
  return <SiteShell><div className="intro"><span className="eyebrow">UK dating pool calculator</span><h1>Who are you<br/><em>looking for?</em></h1><p>Choose your requirements. It can include someone dating a person who lives elsewhere.</p><Select label="Minimum annual income, before tax" value={String(criteria.income)} options={Object.fromEntries(INCOME_OPTIONS.map(v=>[String(v),v===0?"No minimum":"£"+v.toLocaleString("en-GB")+"+"]))} onChange={v=>update({income:Number(v)})}/><fieldset className="height-group"><legend>Height</legend><Select label="Height units" value={unit} options={{cm:"Centimetres",ft:"Feet and inches"}} onChange={setUnit}/><div className="pair"><HeightBound label="Minimum" value={criteria.heightMin} unit={unit} onChange={v=>update({heightMin:v})}/><HeightBound label="Maximum" value={criteria.heightMax} unit={unit} onChange={v=>update({heightMax:v})}/></div><p className="small">Leave a bound blank for no limit. Height shares use a model around published average heights.</p><label>Height spread, cm<input type="number" min="1" max="20" step=".1" value={Number.isFinite(criteria.spread)?criteria.spread:""} onChange={e=>update({spread:e.target.value===""?NaN:Number(e.target.value)})}/></label><p className="small">7 cm is a modelling assumption. Changing it changes the estimate.</p></fieldset><Select label="Sexual orientation" value={criteria.orientation} options={ORIENTATION_LABELS} onChange={v=>update({orientation:v})}/><p className="small">Reported identity; this does not measure attraction to you.</p><Select label="BMI category" value={criteria.bmi} options={BMI_LABELS} onChange={v=>update({bmi:v})}/><Select label="Highest qualification" value={criteria.education} options={EDUCATION_LABELS} onChange={v=>update({education:v})}/><Select label="Ethnic group" value={criteria.ethnicity} options={ETHNICITY_LABELS} onChange={v=>update({ethnicity:v})}/><Select label={childrenStatus.available?"Dependent children living in the household":"Children — data unavailable"} value={criteria.children} options={CHILDREN_LABELS} onChange={v=>update({children:v})} disabled={!childrenStatus.available}/><p className="small">{childrenStatus.explanation}</p><label className="field">Different co-parents<select disabled aria-describedby="coparents-note"><option>Data unavailable</option></select></label><p className="small" id="coparents-note">We cannot account for how many different people someone has children with.</p><button type="button" onClick={()=>{setCriteria(DEFAULT_CRITERIA);setLinkError("");}}>Reset requirements</button></section><div>{calculation.error&&<p role="alert" className="panel error">{calculation.error}</p>}{result&&<><section className="panel result" aria-labelledby="result-heading"><span className="eyebrow">Estimated criteria match</span><h2 id="result-heading" className="result-number">{pct(result.p*100)}</h2><p>of all {SEX_LABELS[result.state.sex].toLowerCase()} aged 18+ in {geographies.find(g=>g.code===result.state.geo)?.name} meet the accounted-for requirements.</p><div className="result-details"><div><strong>≈ {people(result.count)}</strong><span>estimated people</span></div><div><strong>{result.oneIn===null?"No matches in estimate":result.oneIn<=1?"All adults":"1 in "+people(result.oneIn)}</strong><span>in your starting pool</span></div></div><p className="small">Starting pool: {Math.round(result.base).toLocaleString("en-GB")} adults. Published inputs are combined to estimate their overlap.</p><div className="actions"><button onClick={copy}>Copy scenario link</button><button onClick={download}>Download result</button><button disabled={scenarios.length>=3} onClick={()=>{setScenarios(s=>[...s,result.state]);setMessage("Scenario saved for comparison.");}}>Compare ({scenarios.length}/3)</button></div><p className="status" role="status">{message}</p></section><section className="panel"><h2>How your requirements narrow the pool</h2><p className="small">Each percentage remaining uses the same starting pool. Each drop compares with the step directly above.</p>{result.stages.map((s,i)=><div className="funnel-row" key={s.label}><div>{s.label}<div className="bar" aria-hidden="true"><span style={{width:Math.max(0,Math.min(100,s.remainingPct))+"%"}}/></div></div><div className="funnel-values">≈ {people(s.count)}<div className="small">{pct(s.remainingPct)} remain · {pct(s.dropPct)} drop</div>{i===0&&<span className="small">Starting pool</span>}</div></div>)}</section><section className="panel"><h2>If you loosen a requirement</h2><p className="small">These previews use your current location and change only the stated requirement.</p>{result.relaxations.length?result.relaxations.map(r=><div className="relaxation" key={r.label}><div>{r.label}<div className="small">{pct(r.p*100)} · ≈ {people(r.count)} people</div></div><button onClick={()=>update(r.patch)}>Apply</button></div>):<p className="small">No active requirement to loosen.</p>}</section><section className="panel"><h2>Compare your scenarios</h2>{!scenarios.length&&<p className="small">Save up to three sets of requirements using Compare above. Comparisons stay in this tab.</p>}{scenarios.map((c,i)=>{const r=calculatePool(c);return <article className="scenario" key={i}><h3>Scenario {i+1}: {pct(r.p*100)}</h3><p className="small">{SEX_LABELS[c.sex]}, {c.min}–{c.max===90?"90+":c.max}, {geographies.find(g=>g.code===c.geo)?.name} · ≈ {people(r.count)} people</p><details><summary>View all requirements</summary><pre style={{whiteSpace:"pre-wrap"}}>{JSON.stringify(c,null,2)}</pre></details><div className="actions"><button onClick={()=>setCriteria(c)}>Load scenario {i+1}</button><button onClick={()=>setScenarios(s=>s.filter((_,j)=>j!==i))}>Remove scenario {i+1}</button></div></article>;})}</section><section className="panel"><h2>Your result explained</h2><p className="muted">The source links below show where each input comes from. They do not directly count everyone meeting all your requirements together.</p>{result.stages.map(s=>{const source=sourceLibrary.find(x=>x.id===s.sourceId);return <article className="source-proof" key={s.label}><h3>{s.label}</h3><p className="small">≈ {people(s.count)} remain · {pct(s.dropPct)} drop at this step.</p>{source?<><p>{source.explanation}</p><a href={source.url} target="_blank" rel="noopener noreferrer">Read original source ↗</a>{" · "}<Link href={"/sources#"+source.id}>Source details</Link><details><summary>How this input is calculated</summary><p className="small">{String(source.tables)} · {String(source.reference)} · {source.geography}</p>{source.file&&<a href={source.file} download>Download retained data</a>}</details></>:<p className="small">This step uses the published population count.</p>}</article>;})}<h3>What this estimate leaves uncertain</h3><ul className="notes">{result.notes.map(n=><li key={n}>{n}</li>)}</ul>{result.heightSensitivity&&<p className="small">Height model sensitivity: changing the spread gives approximately {people(result.heightSensitivity.low)} to {people(result.heightSensitivity.high)} people. This is a model comparison, not a confidence interval.</p>}<Link href="/methodology">More detail about the calculation</Link></section></>}</div></div></SiteShell>;
}
