// ── Unit helpers ─────────────────────────────────────────────────────────────
export function cmToFeetInches(cm: number): { feet: number; inches: number } {
  const totalInches = cm / 2.54;
  const feet = Math.floor(totalInches / 12);
  const inches = Math.round(totalInches % 12);
  return { feet, inches };
}

export function feetInchesToCm(feet: number, inches: number): number {
  return (feet * 12 + inches) * 2.54;
}

// ── Normal CDF (replaces scipy.stats.norm.cdf) ───────────────────────────────
// erf via Abramowitz & Stegun 7.1.26 (max abs error ~1.5e-7).
function erf(x: number): number {
  const sign = x < 0 ? -1 : 1;
  const ax = Math.abs(x);
  const t = 1 / (1 + 0.3275911 * ax);
  const y =
    1 -
    ((((1.061405429 * t - 1.453152027) * t + 1.421413741) * t - 0.284496736) * t + 0.254829592) *
      t *
      Math.exp(-ax * ax);
  return sign * y;
}

export function normalCdf(x: number, mean: number, std: number): number {
  return 0.5 * (1 + erf((x - mean) / (std * Math.SQRT2)));
}

// ── Individual filter probabilities ──────────────────────────────────────────
export function ageProbability(minAge: number, maxAge: number): number {
  const lo = Math.max(18, Math.min(minAge, maxAge));
  const hi = Math.min(99, Math.max(minAge, maxAge));
  let probability = 0;
  for (const [ageKey, pct] of Object.entries(AGE_SINGLE_YEAR_DISTRIBUTION.value)) {
    if (ageKey === "90+") {
      if (hi >= 90) probability += pct;
      continue;
    }
    const age = Number(ageKey);
    if (age >= lo && age <= hi) probability += pct;
  }
  return probability;
}

function weightedAgeBandValue(
  ageRange: [number, number],
  bands: Array<{ min: number; max: number; mean?: number }>,
  valueForBand: (band: { min: number; max: number; mean?: number }) => number,
): number {
  let years = 0;
  let weighted = 0;
  for (const band of bands) {
    const overlapMin = Math.max(ageRange[0], band.min);
    const overlapMax = Math.min(ageRange[1], band.max);
    if (overlapMax >= overlapMin) {
      const overlapYears = overlapMax - overlapMin + 1;
      years += overlapYears;
      weighted += valueForBand(band) * overlapYears;
    }
  }
  return years > 0 ? weighted / years : valueForBand(bands[0]);
}

function heightParams(gender: Gender, ageRange?: [number, number]): { mean: number; std: number } {
  const params = gender === "Male" ? HEIGHT.male : HEIGHT.female;
  const defaultRange: [number, number] = [18, 99];
  const mean = weightedAgeBandValue(ageRange ?? defaultRange, params.meansByAge, (band) => band.mean ?? 0);
  return { mean, std: params.std };
}

export function heightProbability(minH: number, maxH: number, gender: Gender, ageRange?: [number, number]): number {
  const { mean, std } = heightParams(gender, ageRange);
  return normalCdf(maxH, mean, std) - normalCdf(minH, mean, std);
}

export function getIncomeDistributionForAge(
  gender: Gender,
  ageRange?: [number, number],
): Distribution {
  const byAge =
    gender === "Male" ? INCOME_DISTRIBUTION_BY_AGE_MALE.value : INCOME_DISTRIBUTION_BY_AGE_FEMALE.value;
  if (!ageRange || (ageRange[0] <= 18 && ageRange[1] >= 99)) {
    return gender === "Male" ? INCOME_DISTRIBUTION_MALE.value : INCOME_DISTRIBUTION_FEMALE.value;
  }

  const ageBands: Array<{ key: string; min: number; max: number }> = [
    { key: "18-24", min: 18, max: 24 },
    { key: "25-34", min: 25, max: 34 },
    { key: "35-44", min: 35, max: 44 },
    { key: "45-54", min: 45, max: 54 },
    { key: "55-64", min: 55, max: 64 },
    { key: "65+", min: 65, max: 99 },
  ];

  const lo = Math.max(18, Math.min(ageRange[0], ageRange[1]));
  const hi = Math.min(99, Math.max(ageRange[0], ageRange[1]));
  let totalWeight = 0;
  const weightedDist: Distribution = {};
  for (const b of INCOME_BRACKETS) {
    weightedDist[b.label] = 0;
  }

  for (let age = lo; age <= hi; age++) {
    const ageKey = age >= 90 ? "90+" : String(age);
    const weight = AGE_SINGLE_YEAR_DISTRIBUTION.value[ageKey] ?? 0;
    if (weight <= 0) continue;
    totalWeight += weight;

    const band = ageBands.find((b) => age >= b.min && age <= b.max) ?? ageBands[ageBands.length - 1];
    const distForBand = byAge[band.key] ?? byAge["25-34"];
    for (const b of INCOME_BRACKETS) {
      weightedDist[b.label] += (distForBand[b.label] ?? 0) * weight;
    }
  }

  if (totalWeight <= 0) {
    return gender === "Male" ? INCOME_DISTRIBUTION_MALE.value : INCOME_DISTRIBUTION_FEMALE.value;
  }

  for (const b of INCOME_BRACKETS) {
    weightedDist[b.label] /= totalWeight;
  }
  return weightedDist;
}

export function incomeProbability(
  minIncome: number,
  gender: Gender,
  ageRange?: [number, number],
): number {
  if (minIncome <= 0) return 1.0;
  const dist = getIncomeDistributionForAge(gender, ageRange);
  let probability = 0;
  for (const b of INCOME_BRACKETS) {
    const pct = dist[b.label] ?? 0;
    if (minIncome <= b.low) {
      probability += pct;
    } else if (minIncome < b.high) {
      const width = b.high - b.low;
      const included = b.high - minIncome;
      probability += pct * (included / width);
    }
  }
  return Math.max(0, Math.min(1.0, probability));
}

export function getEducationDistributionForAge(ageRange?: [number, number]): Distribution {
  if (!ageRange || (ageRange[0] <= 18 && ageRange[1] >= 99)) {
    return EDUCATION_DISTRIBUTION.value;
  }

  const ageBands: Array<{ key: string; min: number; max: number }> = [
    { key: "18-24", min: 18, max: 24 },
    { key: "25-34", min: 25, max: 34 },
    { key: "35-44", min: 35, max: 44 },
    { key: "45-54", min: 45, max: 54 },
    { key: "55-64", min: 55, max: 64 },
    { key: "65+", min: 65, max: 99 },
  ];

  const lo = Math.max(18, Math.min(ageRange[0], ageRange[1]));
  const hi = Math.min(99, Math.max(ageRange[0], ageRange[1]));
  let totalWeight = 0;
  const weightedDist: Distribution = {};
  for (const lvl of EDUCATION_ORDER) {
    weightedDist[lvl] = 0;
  }

  for (let age = lo; age <= hi; age++) {
    const ageKey = age >= 90 ? "90+" : String(age);
    const weight = AGE_SINGLE_YEAR_DISTRIBUTION.value[ageKey] ?? 0;
    if (weight <= 0) continue;
    totalWeight += weight;

    const band = ageBands.find((b) => age >= b.min && age <= b.max) ?? ageBands[ageBands.length - 1];
    const distForBand = EDUCATION_DISTRIBUTION_BY_AGE.value[band.key] ?? EDUCATION_DISTRIBUTION.value;
    for (const lvl of EDUCATION_ORDER) {
      weightedDist[lvl] += (distForBand[lvl] ?? 0) * weight;
    }
  }

  if (totalWeight <= 0) return EDUCATION_DISTRIBUTION.value;
  for (const lvl of EDUCATION_ORDER) {
    weightedDist[lvl] /= totalWeight;
  }
  return weightedDist;
}

export function educationProbability(minLevel: string, ageRange?: [number, number]): number {
  if (minLevel === "Any") return 1.0;
  const idx = EDUCATION_ORDER.indexOf(minLevel as (typeof EDUCATION_ORDER)[number]);
  if (idx === -1) return 1.0;
  const dist = getEducationDistributionForAge(ageRange);
  let probability = 0;
  for (let i = idx; i < EDUCATION_ORDER.length; i++) {
    probability += dist[EDUCATION_ORDER[i]] ?? 0;
  }
  return Math.max(0, Math.min(1.0, probability));
}

export function ethnicityProbability(selected: string[]): number {
  return selected.reduce((sum, e) => sum + (ETHNICITY_DISTRIBUTION.value[e] ?? 0), 0);
}

export function bodyTypeProbability(selected: string[], gender: Gender): number {
  const dist = gender === "Male" ? BODY_TYPE_DISTRIBUTION_MALE.value : BODY_TYPE_DISTRIBUTION_FEMALE.value;
  return selected.reduce((sum, b) => sum + (dist[b] ?? 0), 0);
}

export function childrenProbability(acceptable: string[]): number {
  return acceptable.reduce((sum, c) => sum + (CHILDREN_DISTRIBUTION.value[c] ?? 0), 0);
}

function ageBandProbability(ageRange: [number, number], dist: Distribution): number {
  let totalYears = 0;
  let weighted = 0;
  for (const [band, value] of Object.entries(dist)) {
    let lo: number;
    let hi: number;
    if (band.endsWith("+")) {
      lo = Number(band.replace("+", ""));
      hi = 99;
    } else {
      [lo, hi] = band.split("-").map(Number);
    }
    const overlapMin = Math.max(ageRange[0], lo);
    const overlapMax = Math.min(ageRange[1], hi);
    if (overlapMax >= overlapMin) {
      const years = overlapMax - overlapMin + 1;
      totalYears += years;
      weighted += value * years;
    }
  }
  return totalYears > 0 ? weighted / totalYears : 0;
}

export function singleAvailabilityProbability(ageRange: [number, number], gender: Gender): number {
  const dist = gender === "Male" ? SINGLE_AVAILABILITY_BY_AGE.value.male : SINGLE_AVAILABILITY_BY_AGE.value.female;
  return ageBandProbability(ageRange, dist);
}

/**
 * Marriage-history probability. When `mustBeSingle` is true the distribution is
 * renormalised over unmarried people, so it does NOT double-count with the single filter.
 */
export function marriageProbability(
  acceptable: string[],
  userGender: Gender,
  lookingFor: LookingFor,
  orientation: Orientation,
  mustBeSingle = false,
): number {
  const unmarried = ["Never married", "Divorced", "Widowed"];
  const sumOver = (dist: Distribution, keys: string[]) => keys.reduce((s, k) => s + (dist[k] ?? 0), 0);

  const sameSexAttracted =
    (userGender === "Male" && lookingFor === "Male") || (userGender === "Female" && lookingFor === "Female");

  let key: "opposite-sex" | "same-sex" = "opposite-sex";
  if (orientation === "Gay or Lesbian" && sameSexAttracted) key = "same-sex";
  else if (orientation === "Bisexual" && sameSexAttracted) key = "same-sex";
  else if (orientation === "Bisexual" && lookingFor === "Any") {
    // Blend opposite- and same-sex pools.
    const blend = (dist: Distribution) => {
      if (mustBeSingle) {
        const denom = sumOver(dist, unmarried);
        return denom > 0 ? sumOver(dist, acceptable.filter((s) => s !== "Currently married")) / denom : 0;
      }
      return sumOver(dist, acceptable);
    };
    return (blend(MARRIAGE_HISTORY.value["opposite-sex"]) + blend(MARRIAGE_HISTORY.value["same-sex"])) / 2;
  }

  const dist = MARRIAGE_HISTORY.value[key];
  if (mustBeSingle) {
    const denom = sumOver(dist, unmarried);
    return denom > 0 ? sumOver(dist, acceptable.filter((s) => s !== "Currently married")) / denom : 0;
  }
  return sumOver(dist, acceptable);
}

export function baldnessProbability(pref: "Any" | "Not bald" | "Bald or balding", ageRange: [number, number]): number {
  const bands: Array<[number, number, string]> = [
    [18, 29, "18-29"],
    [30, 39, "30-39"],
    [40, 49, "40-49"],
    [50, 59, "50-59"],
    [60, 99, "60+"],
  ];
  let totalYears = 0;
  let weighted = 0;
  for (const [lo, hi, key] of bands) {
    const overlapMin = Math.max(ageRange[0], lo);
    const overlapMax = Math.min(ageRange[1], hi);
    if (overlapMax >= overlapMin) {
      const years = overlapMax - overlapMin + 1;
      totalYears += years;
      weighted += BALDNESS_BY_AGE.value[key] * years;
    }
  }
  const avg = totalYears > 0 ? weighted / totalYears : 0;
  if (pref === "Not bald") return 1 - avg;
  if (pref === "Bald or balding") return avg;
  return 1.0;
}

export function orientationProbability(orientation: Orientation, lookingFor: LookingFor, userGender?: Gender): number {
  const raw = SEXUAL_ORIENTATION_DISTRIBUTION.value;
  const total = raw["Heterosexual/Straight"] + raw["Gay or Lesbian"] + raw["Bisexual"];
  const straight = raw["Heterosexual/Straight"] / total;
  const gay = raw["Gay or Lesbian"] / total;
  const bi = raw["Bisexual"] / total;

  if (lookingFor === "Any") {
    if (orientation === "Bisexual") return straight * 0.5 + gay * 0.5 + bi;
    return bi;
  }
  if (orientation === "Heterosexual/Straight") {
    if (userGender === "Male" && lookingFor === "Male") return bi;
    if (userGender === "Female" && lookingFor === "Female") return bi;
    return straight + bi;
  }
  if (orientation === "Gay or Lesbian") {
    if (userGender === "Male" && lookingFor === "Female") return bi;
    if (userGender === "Female" && lookingFor === "Male") return bi;
    return gay + bi;
  }
  if (orientation === "Bisexual") {
    if (userGender === "Male") return lookingFor === "Male" ? gay + bi : straight + bi;
    if (userGender === "Female") return lookingFor === "Female" ? gay + bi : straight + bi;
    return straight + gay + bi;
  }
  return 1.0;
}

/**
 * Compute regional match distribution across the 12 UK regions.
 * Uses the regional_scale normalization (UK_ADULT_POPULATION / sum(adultPop))
 * so regional adult pools reconcile to the national UK_ADULT_POPULATION.
 */
export function calculateRegionalPool(totalProbability: number): RegionalBreakdownItem[] {
  const entries = Object.entries(UK_REGIONS);
  const totalRegionalAdults = entries.reduce((sum, [, r]) => sum + r.adultPop, 0);
  const regionalScale = totalRegionalAdults > 0 ? UK_ADULT_POPULATION / totalRegionalAdults : 1.0;

  const rawItems = entries.map(([region, r]) => {
    const scaledAdultPop = Math.round(r.adultPop * regionalScale);
    const estimatedMatches = Math.round(r.adultPop * regionalScale * totalProbability);
    const shareOfUkMatches = totalRegionalAdults > 0 ? r.adultPop / totalRegionalAdults : 0;
    return {
      region,
      population: r.population,
      adultPop: r.adultPop,
      scaledAdultPop,
      estimatedMatches,
      shareOfUkMatches,
      lat: r.lat,
      lon: r.lon,
    };
  });

  rawItems.sort((a, b) => b.estimatedMatches - a.estimatedMatches || b.adultPop - a.adultPop);

  const maxAdult = Math.max(...rawItems.map((x) => x.adultPop));
  const minAdult = Math.min(...rawItems.map((x) => x.adultPop));

  return rawItems.map((item, idx) => {
    const densityRatio = maxAdult > minAdult ? (item.adultPop - minAdult) / (maxAdult - minAdult) : 0.5;
    let densityTier: RegionalBreakdownItem["densityTier"] = "Low";
    if (densityRatio >= 0.75) densityTier = "Very High";
    else if (densityRatio >= 0.45) densityTier = "High";
    else if (densityRatio >= 0.25) densityTier = "Medium";

    return {
      rank: idx + 1,
      ...item,
      densityRatio,
      densityTier,
    };
  });
}

function getSelectivityTier(targetPercentage: number, emptyFilters: string[]): SelectivityInfo {
  if (emptyFilters.length > 0) {
    return {
      label: "Zero Pool (Empty Filter)",
      tone: "rose",
      summary: `No matches possible because ${emptyFilters.join(", ")} has zero options selected.`,
    };
  }
  if (targetPercentage < 0.1) {
    return {
      label: "Ultra-Selective (<0.1%)",
      tone: "rose",
      summary: "Needle-in-a-haystack criteria. Relaxing even one strict requirement (e.g. income or height) will multiply your pool.",
    };
  }
  if (targetPercentage < 1.0) {
    return {
      label: "Highly Selective (<1%)",
      tone: "amber",
      summary: "Narrow demographic slice. Fewer than 1 in 100 people in your target gender pool meet all criteria simultaneously.",
    };
  }
  if (targetPercentage < 5.0) {
    return {
      label: "Moderately Selective (1–5%)",
      tone: "violet",
      summary: "Focused dating pool. Selective enough to filter strongly while retaining tens or hundreds of thousands of matches.",
    };
  }
  return {
    label: "Broad Pool (5%+)",
    tone: "emerald",
    summary: "Broad, realistic criteria with a large available matchmaking pool across UK regions.",
  };
}

export function calculateConfidenceInterval(
  estimatedCount: number,
  activeFilterCount: number,
  isSelective: boolean,
): ConfidenceInterval {
  if (estimatedCount <= 0) {
    return {
      low: 0,
      high: 0,
      ciLevel: 0.9,
      relativeMarginPct: 0,
      displayEstimate: "0",
    };
  }

  // Demographic survey error propagation (ONS LFS, HMRC SPI sample, HSE, Census coverage)
  const baseVariance = Math.max(1, activeFilterCount) * 0.0018;
  const tailVariance = isSelective ? 0.0036 : 0.0009;
  const sigma = Math.min(0.4, Math.sqrt(baseVariance + tailVariance));

  const z = 1.644853; // 90% confidence quantile
  const factorLow = Math.exp(-z * sigma);
  const factorHigh = Math.exp(z * sigma);

  const rawLow = Math.round(estimatedCount * factorLow);
  const rawHigh = Math.round(estimatedCount * factorHigh);

  // Demographic rounding helper to eliminate false integer precision
  const roundDemographic = (n: number): number => {
    if (n < 10) return n;
    if (n < 100) return Math.round(n / 5) * 5;
    if (n < 1000) return Math.round(n / 10) * 10;
    if (n < 10000) return Math.round(n / 50) * 50;
    if (n < 100000) return Math.round(n / 100) * 100;
    if (n < 1000000) return Math.round(n / 500) * 500;
    return Math.round(n / 1000) * 1000;
  };

  const roundedPoint = roundDemographic(estimatedCount);
  const low = Math.max(1, roundDemographic(rawLow));
  const high = Math.max(low, roundDemographic(rawHigh));
  const relativeMarginPct = Math.round(((high - low) / (2 * Math.max(1, roundedPoint))) * 100);
  const displayEstimate = `≈ ${roundedPoint.toLocaleString("en-GB")} [${low.toLocaleString("en-GB")} – ${high.toLocaleString("en-GB")}]`;

  return {
    low,
    high,
    ciLevel: 0.9,
    relativeMarginPct,
    displayEstimate,
  };
}

/** Blend a per-gender probability by the population gender split (used when lookingFor === "Any"). */
function blendByGender(fn: (g: Gender) => number): number {
  return GENDER_SPLIT.male * fn("Male") + GENDER_SPLIT.female * fn("Female");
}

export function calculateDatingPool(input: CalculatorInput): CalculatorResult {
  const genderProb =
    input.lookingFor === "Any" ? 1.0 : input.lookingFor === "Male" ? GENDER_SPLIT.male : GENDER_SPLIT.female;
  const basePopulation = Math.round(UK_ADULT_POPULATION * genderProb);
  const baseLabel =
    input.lookingFor === "Any"
      ? "UK adults aged 18+"
      : input.lookingFor === "Male"
      ? "UK adult men aged 18+"
      : "UK adult women aged 18+";

  const ageProb = ageProbability(input.ageRange[0], input.ageRange[1]);

  const heightProb =
    input.lookingFor === "Any"
      ? blendByGender((g) => heightProbability(input.minHeightCm, input.maxHeightCm, g, input.ageRange))
      : heightProbability(input.minHeightCm, input.maxHeightCm, input.lookingFor, input.ageRange);

  const incomeProb =
    input.lookingFor === "Any"
      ? blendByGender((g) => incomeProbability(input.minIncome, g, input.ageRange))
      : incomeProbability(input.minIncome, input.lookingFor, input.ageRange);

  const bodyProb =
    input.lookingFor === "Any"
      ? blendByGender((g) => bodyTypeProbability(input.selectedBodyTypes, g))
      : bodyTypeProbability(input.selectedBodyTypes, input.lookingFor);

  const educationProb = educationProbability(input.educationLevel, input.ageRange);
  const ethnicityProb = ethnicityProbability(input.selectedEthnicities);
  const orientationProb = orientationProbability(input.orientation, input.lookingFor, input.userGender);
  const singleProb = input.mustBeSingle
    ? input.lookingFor === "Any"
      ? blendByGender((g) => singleAvailabilityProbability(input.ageRange, g))
      : singleAvailabilityProbability(input.ageRange, input.lookingFor)
    : 1.0;
  const childrenProb = childrenProbability(input.acceptableChildren);
  const marriageProb = marriageProbability(
    input.acceptableMarriage,
    input.userGender,
    input.lookingFor,
    input.orientation,
    input.mustBeSingle,
  );

  let baldnessProb = 1.0;
  if (input.lookingFor === "Male") {
    baldnessProb = baldnessProbability(input.baldnessPreference, input.ageRange);
  } else if (input.lookingFor === "Any") {
    baldnessProb = GENDER_SPLIT.male * baldnessProbability(input.baldnessPreference, input.ageRange) + GENDER_SPLIT.female * 1.0;
  }

  const emptyFilters: string[] = [];
  if (input.selectedEthnicities.length === 0) emptyFilters.push("Preferred Ethnicity");
  if (input.selectedBodyTypes.length === 0) emptyFilters.push("Preferred Body Type");
  if (input.acceptableChildren.length === 0) emptyFilters.push("Family Status (Children)");
  const effectiveMarriage = input.mustBeSingle
    ? input.acceptableMarriage.filter((m) => m !== "Currently married")
    : input.acceptableMarriage;
  if (effectiveMarriage.length === 0) emptyFilters.push("Acceptable Marital Status");

  const factors: Array<[string, number]> = [
    ["Gender", genderProb],
    ["Age Range", ageProb],
    ["Height Range", heightProb],
    ["Body Type", bodyProb],
    ["Income", incomeProb],
    ["Education", educationProb],
    ["Ethnicity", ethnicityProb],
    ["Orientation", orientationProb],
    ["Single/Available", singleProb],
    ["Children", childrenProb],
    ["Marriage History", marriageProb],
    ["Baldness", baldnessProb],
  ];

  // Identify bottleneck among preference filters (excluding target Gender and single availability)
  const preferenceFactors = factors.filter(([crit]) => crit !== "Gender");
  let bottleneck: BottleneckInfo | null = null;
  if (preferenceFactors.length > 0) {
    const sortedByPassRate = [...preferenceFactors].sort((a, b) => a[1] - b[1]);
    const tightest = sortedByPassRate[0];
    if (tightest && tightest[1] < 1.0) {
      const eliminatedPct = Math.round((1 - tightest[1]) * 100);
      bottleneck = {
        criterion: tightest[0],
        filterPassRate: tightest[1],
        eliminatedPct,
        summary: `${tightest[0]} eliminates ${eliminatedPct}% of potential matches on its own.`,
      };
    }
  }

  // Correlation adjustment:
  // In real populations, high income, advanced qualification (degrees), and age are positively correlated.
  // Independent multiplication assumes independence, which underestimates matches when both are strict.
  // When enabled, we apply an empirical positive-correlation adjustment factor (capped at 2.5x).
  let correlationDiscount = 1.0;
  if (input.correlationAdjustment) {
    const hasHighIncome = input.minIncome >= 50000;
    const hasDegree = input.educationLevel.includes("Undergraduate") || input.educationLevel.includes("Postgraduate");
    if (hasHighIncome && hasDegree) {
      // Statistical joint probability boost factor
      correlationDiscount = input.minIncome >= 100000 ? 1.65 : 1.35;
    } else if (hasHighIncome || hasDegree) {
      correlationDiscount = 1.15;
    }
  }

  let cumulative = 1.0;
  const steps: FilterStep[] = factors.map(([criterion, probability]) => {
    cumulative *= probability;
    return { criterion, probability, remaining: Math.round(UK_ADULT_POPULATION * cumulative) };
  });

  const adjustedCumulative = Math.min(genderProb, cumulative * correlationDiscount);
  const totalProbability = adjustedCumulative;
  const targetProbability = genderProb > 0 ? adjustedCumulative / genderProb : adjustedCumulative;
  const targetPercentage = targetProbability * 100;
  const oneInTargetPool = targetProbability > 0 ? 1 / targetProbability : null;
  const oneInAllAdults = totalProbability > 0 ? 1 / totalProbability : null;
  const selectivityTier = getSelectivityTier(targetPercentage, emptyFilters);
  const regionalBreakdown = calculateRegionalPool(totalProbability);

  // Regional scoping computation
  let scopedPopulation = basePopulation;
  let scopedMatches = Math.round(UK_ADULT_POPULATION * totalProbability);
  let scopedRegionLabel = "All UK";

  if (input.targetRegions && input.targetRegions.length > 0 && !input.targetRegions.includes("All UK")) {
    const selectedRegionalAdults = regionalBreakdown
      .filter((r) => input.targetRegions!.includes(r.region))
      .reduce((sum, r) => sum + r.scaledAdultPop, 0);

    const regionalMatches = regionalBreakdown
      .filter((r) => input.targetRegions!.includes(r.region))
      .reduce((sum, r) => sum + r.estimatedMatches, 0);

    scopedPopulation = Math.round(selectedRegionalAdults * genderProb);
    scopedMatches = regionalMatches;
    scopedRegionLabel =
      input.targetRegions.length === 1
        ? input.targetRegions[0]
        : `${input.targetRegions.length} Regions (${input.targetRegions.join(", ")})`;
  }

  const estimatedMatches = Math.round(UK_ADULT_POPULATION * totalProbability);
  const activeFilterCount = factors.filter(([, p]) => p < 1.0).length;
  const isSelective = targetPercentage < 1.0;
  const confidenceInterval = calculateConfidenceInterval(estimatedMatches, activeFilterCount, isSelective);
  const scopedCi = calculateConfidenceInterval(scopedMatches, activeFilterCount, isSelective);
  const scopedMatchesLow = scopedCi.low;
  const scopedMatchesHigh = scopedCi.high;

  // Sensitivity & Trade-Offs computation
  let tradeOffs: SensitivityTradeOff[] = [];
  if (!input.skipSensitivity) {
    tradeOffs = calculateSensitivityTradeOffs(input, estimatedMatches);
  }

  return {
    totalProbability,
    percentage: totalProbability * 100,
    targetPercentage,
    basePopulation,
    baseLabel,
    estimatedMatches,
    confidenceInterval,
    oneInTargetPool,
    oneInAllAdults,
    selectivityTier,
    emptyFilters,
    bottleneck,
    correlationDiscount,
    regionalBreakdown,
    steps,
    scopedPopulation,
    scopedMatches,
    scopedMatchesLow,
    scopedMatchesHigh,
    scopedRegionLabel,
    tradeOffs,
  };
}

const INCOME_TIERS = [0, 11440, 25000, 37430, 42200, 50000, 75000, 100000, 150000, 250000, 1000000];

export function calculateSensitivityTradeOffs(
  input: CalculatorInput,
  currentMatches: number,
): SensitivityTradeOff[] {
  const tradeOffs: SensitivityTradeOff[] = [];
  const baseInput = { ...input, skipSensitivity: true };

  // 1. Income relaxation
  if (input.minIncome > 0) {
    const lowerTiers = INCOME_TIERS.filter((t) => t < input.minIncome);
    const lowerTier = lowerTiers.length > 0 ? lowerTiers[lowerTiers.length - 1] : 0;
    const relaxed = calculateDatingPool({ ...baseInput, minIncome: lowerTier });
    const gainMatches = relaxed.estimatedMatches - currentMatches;
    if (gainMatches > 0) {
      const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
      const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
      tradeOffs.push({
        filter: "Income",
        relaxation:
          lowerTier === 0
            ? "Remove income requirement (Any income)"
            : `Lower min salary to £${lowerTier.toLocaleString("en-GB")}+`,
        currentMatches,
        newMatches: relaxed.estimatedMatches,
        gainMatches,
        multiplier,
        gainPercentage,
      });
    }
  }

  // 2. Height relaxation
  if (input.minHeightCm > 150 || input.maxHeightCm < 200) {
    const newMin = Math.max(140, input.minHeightCm - 5);
    const newMax = Math.min(210, input.maxHeightCm + 5);
    const relaxed = calculateDatingPool({ ...baseInput, minHeightCm: newMin, maxHeightCm: newMax });
    const gainMatches = relaxed.estimatedMatches - currentMatches;
    if (gainMatches > 0) {
      const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
      const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
      tradeOffs.push({
        filter: "Height Range",
        relaxation: `Relax height boundary by ±5 cm (to ${newMin}–${newMax} cm)`,
        currentMatches,
        newMatches: relaxed.estimatedMatches,
        gainMatches,
        multiplier,
        gainPercentage,
      });
    }
  }

  // 3. Age range expansion (±2 years)
  if (input.ageRange[1] - input.ageRange[0] < 40) {
    const newAgeMin = Math.max(18, input.ageRange[0] - 2);
    const newAgeMax = Math.min(99, input.ageRange[1] + 2);
    if (newAgeMin !== input.ageRange[0] || newAgeMax !== input.ageRange[1]) {
      const relaxed = calculateDatingPool({ ...baseInput, ageRange: [newAgeMin, newAgeMax] });
      const gainMatches = relaxed.estimatedMatches - currentMatches;
      if (gainMatches > 0) {
        const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
        const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
        tradeOffs.push({
          filter: "Age Range",
          relaxation: `Widen age bracket by ±2 years (${newAgeMin}–${newAgeMax})`,
          currentMatches,
          newMatches: relaxed.estimatedMatches,
          gainMatches,
          multiplier,
          gainPercentage,
        });
      }
    }
  }

  // 4. Education relaxation
  if (input.educationLevel !== "Any") {
    const relaxed = calculateDatingPool({ ...baseInput, educationLevel: "Any" });
    const gainMatches = relaxed.estimatedMatches - currentMatches;
    if (gainMatches > 0) {
      const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
      const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
      tradeOffs.push({
        filter: "Education",
        relaxation: "Accept any education level (remove degree requirement)",
        currentMatches,
        newMatches: relaxed.estimatedMatches,
        gainMatches,
        multiplier,
        gainPercentage,
      });
    }
  }

  // 5. Single / Availability relaxation
  if (input.mustBeSingle) {
    const relaxed = calculateDatingPool({ ...baseInput, mustBeSingle: false });
    const gainMatches = relaxed.estimatedMatches - currentMatches;
    if (gainMatches > 0) {
      const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
      const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
      tradeOffs.push({
        filter: "Single Status",
        relaxation: "Include all relationship/cohabitation statuses",
        currentMatches,
        newMatches: relaxed.estimatedMatches,
        gainMatches,
        multiplier,
        gainPercentage,
      });
    }
  }

  // 6. Body type relaxation
  const allBodyKeys = Object.keys(
    input.lookingFor === "Male"
      ? BODY_TYPE_DISTRIBUTION_MALE.value
      : BODY_TYPE_DISTRIBUTION_FEMALE.value
  );
  if (input.selectedBodyTypes.length < allBodyKeys.length && input.selectedBodyTypes.length > 0) {
    const relaxed = calculateDatingPool({ ...baseInput, selectedBodyTypes: allBodyKeys });
    const gainMatches = relaxed.estimatedMatches - currentMatches;
    if (gainMatches > 0) {
      const multiplier = currentMatches > 0 ? Number((relaxed.estimatedMatches / currentMatches).toFixed(2)) : 1;
      const gainPercentage = currentMatches > 0 ? Math.round((gainMatches / currentMatches) * 100) : 100;
      tradeOffs.push({
        filter: "Body Type",
        relaxation: "Accept all body types",
        currentMatches,
        newMatches: relaxed.estimatedMatches,
        gainMatches,
        multiplier,
        gainPercentage,
      });
    }
  }

  // Sort by highest multiplier / match gain
  tradeOffs.sort((a, b) => b.multiplier - a.multiplier || b.gainMatches - a.gainMatches);

  return tradeOffs.slice(0, 4);
}
