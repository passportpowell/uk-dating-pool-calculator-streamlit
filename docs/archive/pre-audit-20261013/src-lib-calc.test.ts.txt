function expect(actual: any) {
  return {
    toBeCloseTo: (expected: number, precision: number = 2) => {
      const tolerance = Math.pow(10, -precision) * 1.000001;
      assert.ok(
        Math.abs(actual - expected) <= tolerance,
        `Expected ${actual} to be close to ${expected} with precision ${precision}`
      );
    },
    toBe: (expected: any) => assert.strictEqual(actual, expected),
    toBeGreaterThan: (expected: number) => assert.ok(actual > expected, `Expected ${actual} > ${expected}`),
    toBeLessThan: (expected: number) => assert.ok(actual < expected, `Expected ${actual} < ${expected}`),
    toBeGreaterThanOrEqual: (expected: number) => assert.ok(actual >= expected, `Expected ${actual} >= ${expected}`),
    toBeLessThanOrEqual: (expected: number) => assert.ok(actual <= expected, `Expected ${actual} <= ${expected}`),
    not: {
      toBeNull: () => assert.notStrictEqual(actual, null),
      toBe: (expected: any) => assert.notStrictEqual(actual, expected),
    },
  };
}

const sum = (d: Distribution) => Object.values(d).reduce((a, b) => a + b, 0);

describe("distributions sum to 1.0", () => {
  const cases: Array<[string, Distribution]> = [
    ["age", AGE_DISTRIBUTION.value],
    ["ethnicity", ETHNICITY_DISTRIBUTION.value],
    ["income male", INCOME_DISTRIBUTION_MALE.value],
    ["income female", INCOME_DISTRIBUTION_FEMALE.value],
    ["education", EDUCATION_DISTRIBUTION.value],
    ["body male", BODY_TYPE_DISTRIBUTION_MALE.value],

    ["marriage opposite-sex", MARRIAGE_HISTORY.value["opposite-sex"]],
    ["marriage same-sex", MARRIAGE_HISTORY.value["same-sex"]],

  for (const [name, dist] of cases) {
    it(name, () => {
      expect(sum(dist)).toBeCloseTo(1.0, 6);
    });
  }

  it("orientation distribution sums to 1.0", () => {
    expect(sum(SEXUAL_ORIENTATION_DISTRIBUTION.value)).toBeCloseTo(1.0, 6);
  });
});

describe("regional pool breakdown & reconciliation", () => {
  it("returns 12 ranked UK regions whose scaled matches reconcile to national total", () => {
    const p = 0.01;
    const regions = calculateRegionalPool(p);
    expect(regions.length).toBe(Object.keys(UK_REGIONS).length);
    expect(regions[0].rank).toBe(1);
    expect(regions[0].region).toBe("South East");
    expect(regions[1].region).toBe("London");

    const totalShare = regions.reduce((acc, r) => acc + r.shareOfUkMatches, 0);
    expect(totalShare).toBeCloseTo(1.0, 6);

    const sumRegionalMatches = regions.reduce((acc, r) => acc + r.estimatedMatches, 0);
    const nationalExpected = Math.round(UK_ADULT_POPULATION * p);
    // Rounding across 12 regions can differ by at most 6 people
    assert.ok(Math.abs(sumRegionalMatches - nationalExpected) <= 12);
  });
});

describe("sensitivity and trade-offs analysis", () => {
  it("generates actionable relaxation recommendations with positive multipliers", () => {
    const strictInput = {
      userGender: "Female" as const,
      orientation: "Heterosexual/Straight" as const,
      lookingFor: "Male" as const,
      ageRange: [28, 35] as [number, number],
      minHeightCm: 183, // 6'0"+
      maxHeightCm: 210,
      minIncome: 75000,
      educationLevel: "Undergraduate degree",
      selectedEthnicities: Object.keys(ETHNICITY_DISTRIBUTION.value),
      selectedBodyTypes: ["Healthy weight (BMI 18.5-24.9)"],
      mustBeSingle: true,
      acceptableChildren: ["No children"],
      acceptableMarriage: ["Never married"],
      baldnessPreference: "Not bald" as const,
    };

    const res = calculateDatingPool(strictInput);
    expect(res.tradeOffs.length).toBeGreaterThan(0);

    const incomeTradeOff = res.tradeOffs.find((t) => t.filter === "Income");
    expect(incomeTradeOff).not.toBeNull();
    if (incomeTradeOff) {
      expect(incomeTradeOff.multiplier).toBeGreaterThan(1.0);
      expect(incomeTradeOff.gainMatches).toBeGreaterThan(0);
      expect(incomeTradeOff.newMatches).toBeGreaterThan(incomeTradeOff.currentMatches);
    }
  });
});

describe("bivariate age-income conditioning eliminates young high-earner distortion", () => {
  it("young male cohort (18-24) earning £75k+ is under 0.3% (eliminating 20x overcount)", () => {
    const youngProb = incomeProbability(75000, "Male", [18, 24]);
    expect(youngProb).toBeLessThan(0.003); // < 0.3%
    expect(youngProb).toBeGreaterThan(0.001); // > 0.1%
  });

  it("prime earning male cohort (45-54) earning £75k+ reflects peak earnings (>10%)", () => {
    const primeProb = incomeProbability(75000, "Male", [45, 54]);
    expect(primeProb).toBeGreaterThan(0.12); // > 12%
    expect(primeProb).toBeLessThan(0.20); // < 20%
  });

  it("full adult age range (18-99) matches national marginal distribution", () => {
    const nationalProb = incomeProbability(75000, "Male");
    const conditionedAllAdults = incomeProbability(75000, "Male", [18, 99]);
    expect(conditionedAllAdults).toBeCloseTo(nationalProb, 3);
  });
});

describe("confidence intervals and demographic error propagation", () => {
  it("generates strictly positive 90% error bounds where low <= estimatedMatches <= high", () => {
    const ci = calculateConfidenceInterval(10000, 5, false);
    expect(ci.low).toBeGreaterThan(0);
    expect(ci.high).toBeGreaterThan(ci.low);
    expect(ci.low).toBeLessThanOrEqual(10000);

    expect(ci.relativeMarginPct).toBeGreaterThan(5);
    expect(ci.relativeMarginPct).toBeLessThan(40);
